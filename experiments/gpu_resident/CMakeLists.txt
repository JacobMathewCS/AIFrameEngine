cmake_minimum_required(VERSION 3.25)
cmake_policy(SET CMP0141 NEW)
set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT Embedded)
project(GpuResidentExperiment LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This experiment requires Windows x64 and MSVC.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This experiment requires an x64 build.")
endif()
set(AIFRAME_ORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../external/onnxruntime-directml"
    CACHE PATH "Matching ONNX Runtime DirectML include/ and lib/ distribution")
foreach(required include/onnxruntime_cxx_api.h include/dml_provider_factory.h
        lib/onnxruntime.lib lib/onnxruntime.dll lib/DirectML.dll)
    if(NOT EXISTS "${AIFRAME_ORT_ROOT}/${required}")
        message(FATAL_ERROR "Missing ${required}; set AIFRAME_ORT_ROOT to a matching DirectML SDK.")
    endif()
endforeach()
add_library(GpuResidentExperiment STATIC interop/gpu_bound_session.cpp compile_check.cpp)
target_compile_features(GpuResidentExperiment PUBLIC cxx_std_17)
target_compile_definitions(GpuResidentExperiment PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(GpuResidentExperiment PRIVATE /W4 /WX /EHsc /fp:precise)
target_include_directories(GpuResidentExperiment SYSTEM PUBLIC "${AIFRAME_ORT_ROOT}/include")
target_link_libraries(GpuResidentExperiment PUBLIC d3d11.lib d3d12.lib dxgi.lib d3dcompiler.lib
    "${AIFRAME_ORT_ROOT}/lib/onnxruntime.lib")

include(CTest)
option(AIFRAME_GPU_HARDWARE_SAFETY "Enable synthetic safety tests on NVIDIA 10de:2520" OFF)
if(BUILD_TESTING)
    # Compile separately: the oracle macro changes class definitions, so the test
    # must not link objects compiled without that macro from the isolated library.
    add_executable(CombinedTest combined_test.cpp interop/gpu_bound_session.cpp)
    target_compile_features(CombinedTest PRIVATE cxx_std_17)
    target_compile_definitions(CombinedTest PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN AIFRAME_COMBINED_TEST_ORACLES)
    target_compile_options(CombinedTest PRIVATE /W4 /WX /wd4505 /EHsc /fp:precise)
    target_include_directories(CombinedTest SYSTEM PRIVATE "${AIFRAME_ORT_ROOT}/include")
    target_link_libraries(CombinedTest PRIVATE d3d11.lib d3d12.lib dxgi.lib d3dcompiler.lib
        bcrypt.lib "${AIFRAME_ORT_ROOT}/lib/onnxruntime.lib")
    foreach(dll onnxruntime.dll DirectML.dll onnxruntime_providers_shared.dll)
        if(EXISTS "${AIFRAME_ORT_ROOT}/lib/${dll}")
            add_custom_command(TARGET CombinedTest POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${AIFRAME_ORT_ROOT}/lib/${dll}" "$<TARGET_FILE_DIR:CombinedTest>/${dll}")
        endif()
    endforeach()
    add_test(NAME WarpSafety COMMAND CombinedTest --warp-no-inference)
    set_tests_properties(WarpSafety PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
    if(AIFRAME_GPU_HARDWARE_SAFETY)
        add_test(NAME HardwareSafety COMMAND CombinedTest --hardware-no-inference)
        set_tests_properties(HardwareSafety PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
    endif()
    add_test(NAME RejectMissing COMMAND CombinedTest)
    set_tests_properties(RejectMissing PROPERTIES WILL_FAIL TRUE TIMEOUT 10)
    # Model inference is never registered with CTest.
endif()
